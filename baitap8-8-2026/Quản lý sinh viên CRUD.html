from flask import Flask, request, redirect, render_template_string import sqlite3 app = Flask(__name__) DATABASE = "students.db" # ========================================================= # DATABASE # ========================================================= def get_db(): conn = sqlite3.connect(DATABASE) conn.row_factory = sqlite3.Row return conn def init_db(): conn = get_db() conn.execute(""" CREATE TABLE IF NOT EXISTS students ( id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT NOT NULL, age INTEGER NOT NULL, email TEXT NOT NULL, class_name TEXT NOT NULL ) """) conn.commit() conn.close() # ========================================================= # HTML + CSS + JAVASCRIPT # ========================================================= HTML = """ Quản lý sinh viên

🎓 Quản lý sinh viên

➕ Thêm sinh viên

🔍 Tìm kiếm

📋 Danh sách sinh viên ({{ students|length }})

{% for student in students %} {% else %} {% endfor %}
ID Họ tên Tuổi Email Lớp Thao tác
{{ student["id"] }} {{ student["name"] }} {{ student["age"] }} {{ student["email"] }} {{ student["class_name"] }}
Chưa có sinh viên nào.

✏️ Chỉnh sửa sinh viên

""" # ========================================================= # HOME # ========================================================= @app.route("/") def index(): keyword = request.args.get( "keyword", "" ).strip() conn = get_db() if keyword: students = conn.execute(""" SELECT * FROM students WHERE name LIKE ? OR email LIKE ? OR class_name LIKE ? ORDER BY id DESC """, ( f"%{keyword}%", f"%{keyword}%", f"%{keyword}%" )).fetchall() else: students = conn.execute(""" SELECT * FROM students ORDER BY id DESC """).fetchall() conn.close() return render_template_string( HTML, students=students, keyword=keyword ) # ========================================================= # ADD # ========================================================= @app.route("/add", methods=["POST"]) def add_student(): name = request.form.get( "name", "" ).strip() age = request.form.get( "age", "" ) email = request.form.get( "email", "" ).strip() class_name = request.form.get( "class_name", "" ).strip() if name and age and email and class_name: conn = get_db() conn.execute(""" INSERT INTO students (name, age, email, class_name) VALUES (?, ?, ?, ?) """, ( name, age, email, class_name )) conn.commit() conn.close() return redirect("/") # ========================================================= # EDIT # ========================================================= @app.route("/edit/", methods=["POST"]) def edit_student(id): name = request.form.get( "name", "" ).strip() age = request.form.get( "age", "" ) email = request.form.get( "email", "" ).strip() class_name = request.form.get( "class_name", "" ).strip() if name and age and email and class_name: conn = get_db() conn.execute(""" UPDATE students SET name = ?, age = ?, email = ?, class_name = ? WHERE id = ? """, ( name, age, email, class_name, id )) conn.commit() conn.close() return redirect("/") # ========================================================= # DELETE # ========================================================= @app.route("/delete/", methods=["POST"]) def delete_student(id): conn = get_db() conn.execute( "DELETE FROM students WHERE id = ?", (id,) ) conn.commit() conn.close() return redirect("/") # ========================================================= # START # ========================================================= if __name__ == "__main__": init_db() print("=" * 50) print("QUAN LY SINH VIEN") print("=" * 50) print("Website: http://127.0.0.1:5000") print("=" * 50) app.run(debug=True)
